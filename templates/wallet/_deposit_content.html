function depositWizard() { return { step: 1, amount: null, currency: '{{ account.currency if account else "UGX" }}', baseCurrency: '{{ account.currency if account else "UGX" }}', limit: {{ limits.daily_limit }}, perTxnDepositLimit: {{ limits.per_transaction.deposit }}, sourceOfFunds: '', method: '', fxRate: null, fxLoading: false, fxError: null, init() { this.$watch('currency', () => this.updateFxQuote()); this.$watch('amount', () => this.updateFxQuote()); }, async updateFxQuote() { if (!this.amount || this.currency === this.baseCurrency) { this.fxRate = null; this.fxError = null; return; } // Only fetch rate if we don't have it or it's a different currency pair this.fxLoading = true; this.fxError = null; try { const response = await fetch(`/api/fx/rates/${this.currency}/${this.baseCurrency}`); const data = await response.json(); if (response.ok) { this.fxRate = data.data.rate; } else { this.fxError = data.error || 'Failed to fetch exchange rate'; this.fxRate = null; } } catch (error) { this.fxError = 'Network error fetching exchange rate'; this.fxRate = null; } finally { this.fxLoading = false; } }, get convertedAmount() { if (!this.amount) return null; if (this.currency === this.baseCurrency) return this.amount; if (!this.fxRate) return null; return (this.amount * this.fxRate).toFixed(2); }, formatCurrency(val, curr) { if(!val) return '0.00'; return new Intl.NumberFormat('en-US', { style: 'currency', currency: curr }).format(val); }, nextStep(currentStep) { if (currentStep === 1) { let checkAmount = this.convertedAmount || this.amount; if (checkAmount > 10000000) { this.step = 2; // Require Source of Funds } else { this.step = 3; // Skip directly to payment method } } else if (currentStep === 2) { this.step = 3; } } } }
